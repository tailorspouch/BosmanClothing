rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null && request.auth.token.email != null; }
    function mail() { return request.auth.token.email.lower(); }
    function isAdmin() { return signedIn() && mail() in ['thesamueloseni@gmail.com']; }
    function isOwner() { return signedIn() && mail() in ['ogunjobitobitim@gmail.com']; }
    function isStaff() {
      return signedIn() && (isAdmin() || isOwner() ||
        exists(/databases/$(database)/documents/staff/$(mail())));
    }
    match /staff/{email} {
      allow read: if isStaff();
      allow create, update, delete: if isAdmin() || isOwner();
    }
    match /clients/{id} {
      allow read: if isStaff();
      allow create: if isStaff() && request.resource.data.id == id && request.resource.data.name is string && request.resource.data.updated is string;
      allow update: if isStaff() && request.resource.data.id == id && request.resource.data.updated is string && request.resource.data.updated >= resource.data.updated;
      allow delete: if false;
    }
    match /photos/{id} {
      allow read: if isStaff();
      allow create: if isStaff() && request.resource.data.d is string && request.resource.data.d.size() < 1000000;
      allow update, delete: if false;
    }
    match /meta/{doc} {
      allow read: if isStaff();
      allow create: if isAdmin() || isOwner();
      allow update: if isAdmin() || isOwner() || (isStaff() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastWeek', 'lastMonth']));
      allow delete: if false;
    }
    match /trash/{id} {
      allow read: if isStaff();
      allow create: if isStaff();
      allow update: if (isAdmin() || isOwner()) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['hide']);
      allow delete: if false;
    }
    match /tasks/{id} {
      allow read: if isAdmin() || isOwner() || (isStaff() && (resource.data.to == mail() || resource.data.to == 'all'));
      allow create, delete: if isAdmin() || isOwner();
      allow update: if isAdmin() || isOwner() || (isStaff() && (resource.data.to == mail() || resource.data.to == 'all') && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['done', 'doneBy', 'doneAt']));
    }
    match /todos/{id} { allow read, create, update, delete: if isStaff(); }
    match /admintodos/{id} { allow read, create, update, delete: if isAdmin() || isOwner(); }
  }
}
